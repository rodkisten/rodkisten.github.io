// Photo Manager 4.8.0 · sequential loader, Vue Router 4.6.3 pinned.
// Load both scripts in order. If CDNs are offline on a first visit,
// leave the previous controller usable until a network connection returns.
(() => {
  "use strict";
  const v = "4.6.3";
  const sources = [
    `https://cdn.jsdelivr.net/npm/vue-router@${v}/dist/vue-router.global.prod.js`,
    `https://unpkg.com/vue-router@${v}/dist/vue-router.global.prod.js`,
  ];
  function load(url) {
    return new Promise((resolve, reject) => {
      const script = document.createElement("script");
      script.src = url;
      script.async = false;
      script.onload = resolve;
      script.onerror = () => {
        script.remove();
        reject(new Error(`Failed to load ${url}`));
      };
      document.body.appendChild(script);
    });
  }
  async function start() {
    if (!window.VueRouter) {
      for (const source of sources) {
        try { await load(source); if (window.VueRouter) break; }
        catch { /* try the pinned mirror */ }
      }
    }
    await load("./assets/app.js");
    await load("./assets/ui.js");
  }
  start().catch(error => {
    const banner = document.createElement("p");
    banner.className = "vue-loading-error";
    banner.role = "alert";
    banner.textContent = `A interface não carregou: ${error.message}`;
    document.body.prepend(banner);
  });
})();
